### /home/ubuntu/cakna/deploy/docker-compose.yml ###
# ─────────────────────────────────────────────────────────────────────────────
# Cakna — single-instance production stack: PostgreSQL + Node app + Caddy (TLS).
# Run from this directory:   docker compose up -d --build
# Values come from ./.env  (copy .env.example -> .env and fill it in).
# ─────────────────────────────────────────────────────────────────────────────
name: cakna

services:
  # PostgreSQL 16 — data persists in the named volume `pgdata`.
  postgres:
    image: postgres:16
    environment:
      POSTGRES_USER: cakna
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:?set POSTGRES_PASSWORD in deploy/.env}
      POSTGRES_DB: cakna
    volumes:
      - pgdata:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U cakna -d cakna"]
      interval: 5s
      timeout: 5s
      retries: 10
    restart: unless-stopped

  # The Node app. Built from the repo root (context: ..) so the Dockerfile can
  # COPY server/ build/ pwa/. Not published to the host — Caddy fronts it.
  app:
    build:
      context: ..
      dockerfile: deploy/Dockerfile
    env_file:
      - .env                       # QCXIS_*, ADMIN_EMAILS, SEED_DEMO, COOKIE_SECURE, ...
    environment:
      # DB host = the `postgres` service name on the compose network.
      DATABASE_URL: postgres://cakna:${POSTGRES_PASSWORD}@postgres:5432/cakna
      # Secure cookies: 1 for HTTPS (default). Override to 0 in .env ONLY for
      # the HTTP/IP-only Caddy ":80" testing mode.
      COOKIE_SECURE: "${COOKIE_SECURE:-1}"
      # Internal port Caddy proxies to (keep 3000 to match the Caddyfile).
      PORT: "3000"
    expose:
      - "3000"                     # internal only; NOT published to the host
    depends_on:
      postgres:
        condition: service_healthy
    restart: unless-stopped

  # Caddy — terminates TLS and reverse-proxies to the app. Auto-provisions a
  # Let's Encrypt certificate for $DOMAIN (needs ports 80+443 reachable + DNS).
  caddy:
    image: caddy:2
    ports:
      - "80:80"
      - "443:443"
    environment:
      DOMAIN: ${DOMAIN:?set DOMAIN in deploy/.env}   # used by {$DOMAIN} in the Caddyfile
    volumes:
      - ./Caddyfile:/etc/caddy/Caddyfile:ro
      - caddy_data:/data           # persists issued certificates
      - caddy_config:/config
